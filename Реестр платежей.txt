// === СОЗДАНИЕ ЗАЯВКИ ===
function createRequest(formData, files) {
  try {
    Logger.log('=== НАЧАЛО СОЗДАНИЯ ЗАЯВКИ ===');
    var userEmail = Session.getActiveUser().getEmail();
    if (!userEmail) {
      throw new Error('Пользователь не определен. Выполните авторизацию.');
    }
    var ss = SpreadsheetApp.getActiveSpreadsheet();
    var sheet = ss.getSheetByName(НАСТРОЙКИ.листРеестр);
    if (!sheet) {
      sheet = ss.insertSheet(НАСТРОЙКИ.листРеестр);
      initializeNewSheet(sheet);
    }
    var lastRow = sheet.getLastRow();
    var requestNumber = 'ЗАЯВКА-' + new Date().getFullYear() + '-' + 
                        String(lastRow).padStart(4, '0');
    var folder = null;
    var folderLink = '';
    var fileLinks = [];
    try {
      folder = createRequestFolder(requestNumber, formData, userEmail);
      if (folder) {
        folderLink = folder.getUrl();
        if (files && files.length > 0) {
          fileLinks = uploadFilesToFolder(folder, files);
        }
      }
    } catch (folderError) {
      Logger.log('Ошибка папки: ' + folderError);
    }
    var newRow = lastRow + 1;
    sheet.insertRowAfter(lastRow);
    sheet.getRange(newRow, 1).setValue(new Date(formData.dateRequest));
    sheet.getRange(newRow, 2).setValue(new Date(formData.datePayment));
    sheet.getRange(newRow, 3).setValue(formData.initiator || '');
    sheet.getRange(newRow, 4).setValue(formData.department || '');
    sheet.getRange(newRow, 5).setValue(formData.counterparty || '');
    sheet.getRange(newRow, 6).setValue(formData.purpose || '');
    sheet.getRange(newRow, 7).setValue(parseFloat(formData.amount) || 0);
    sheet.getRange(newRow, 8).setValue(formData.currency || 'RUB');
    if (fileLinks.length > 0) {
      sheet.getRange(newRow, 9).setValue(fileLinks.join('\n'));
    }
    sheet.getRange(newRow, 10).setValue("На согласовании");
    sheet.getRange(newRow, 14).setValue("Не оплачено");
    sheet.getRange(newRow, 17).setValue(userEmail);
    sheet.getRange(newRow, 18).setValue(requestNumber);
    if (folderLink) {
      sheet.getRange(newRow, 19).setValue(folderLink);
    }
    addRubFormula(newRow);
    Logger.log('=== ЗАЯВКА СОЗДАНА: ' + requestNumber + ' ===');
    return requestNumber;
  } catch (error) {
    Logger.log('ОШИБКА: ' + error.toString());
    throw new Error('Ошибка: ' + error.toString());
  }
}

// === ПЕРЕСЧЕТ В РУБЛИ ===
function addRubFormula(row) {
  var sheet = SpreadsheetApp.getActive().getSheetByName(НАСТРОЙКИ.листРеестр);
  var валюта = sheet.getRange(row, НАСТРОЙКИ.колонки.валюта).getValue();
  var формула = '=G' + row;
  if (валюта == "USD" || валюта == "EUR" || валюта == "CNY") {
    формула = '=G' + row + '*GOOGLEFINANCE("CURRENCY:' + валюта + 'RUB")';
  }
  sheet.getRange(row, НАСТРОЙКИ.колонки.суммаРуб).setFormula(формула);
}

// === ОБНОВЛЕНИЕ КУРСОВ ===
function refreshCurrencyRates() {
  var sheet = SpreadsheetApp.getActive().getSheetByName(НАСТРОЙКИ.листРеестр);
  var lastRow = sheet.getLastRow();
  for (var row = 2; row <= lastRow; row++) {
    try { addRubFormula(row); } catch(e) {}
  }
  showAlert('✅ Курсы обновлены');
}

// === СОЗДАНИЕ ПАПКИ ===
function createRequestFolder(requestNumber, formData, userEmail) {
  try {
    var rootFolder;
    if (НАСТРОЙКИ.корневаяПапка && НАСТРОЙКИ.корневаяПапка.trim() !== '') {
      try {
        rootFolder = DriveApp.getFolderById(НАСТРОЙКИ.корневаяПапка.trim());
      } catch (e) {
        rootFolder = DriveApp.getRootFolder();
      }
    } else {
      rootFolder = DriveApp.getRootFolder();
    }
    var safeCounterparty = (formData.counterparty || 'Контрагент').replace(/[^\w\sа-яА-ЯёЁ]/gi, '');
    var folder = rootFolder.createFolder(requestNumber + ' - ' + safeCounterparty);
    folder.createFile('описание.txt', 'Заявка: ' + requestNumber + '\nСумма: ' + formData.amount + ' ' + formData.currency);
    return folder;
  } catch (error) {
    return null;
  }
}

// === ЗАГРУЗКА ФАЙЛОВ ===
function uploadFilesToFolder(folder, files) {
  var fileLinks = [];
  if (!folder) return fileLinks;
  for (var i = 0; i < files.length; i++) {
    try {
      var blob = Utilities.newBlob(Utilities.base64Decode(files[i].data), files[i].type, files[i].name);
      fileLinks.push(folder.createFile(blob).getUrl());
    } catch (error) {}
  }
  return fileLinks;
}

// === СОГЛАСОВАНИЕ ЗАЯВКИ ===
function approveRequest(row, comment) {

    var userEmail = Session.getActiveUser().getEmail();

  } catch (error) {
    return { success: false, message: error.toString() };
  }

// === СНЯТИЕ ВСЕХ ЗАЩИТ ===
function removeAllProtectionsCompletely() {
  var ss = SpreadsheetApp.getActiveSpreadsheet();
  var sheets = ss.getSheets();
  for (var s = 0; s < sheets.length; s++) {
    var sheet = sheets[s];
    var protections = sheet.getProtections(SpreadsheetApp.ProtectionType.SHEET);
    for (var i = 0; i < protections.length; i++) protections[i].remove();
    var rangeProtections = sheet.getProtections(SpreadsheetApp.ProtectionType.RANGE);
    for (var i = 0; i < rangeProtections.length; i++) rangeProtections[i].remove();
  }
  showAlert('✅ Защиты сняты');
}

// === УСЛОВНОЕ ФОРМАТИРОВАНИЕ ===
function setupConditionalFormatting() {
  var sheet = SpreadsheetApp.getActive().getSheetByName(НАСТРОЙКИ.листРеестр);
  if (!sheet) return;
  sheet.clearConditionalFormatRules();
  var rules = [];
  rules.push(SpreadsheetApp.newConditionalFormatRule().setRanges([sheet.getRange("A2:S")]).setBackground("#ffcdd2").whenFormulaSatisfied('=OR($J2="Отклонено", AND($J2="Согласовано", $N2<>"Оплачено", $B2<TODAY()))').build());
  rules.push(SpreadsheetApp.newConditionalFormatRule().setRanges([sheet.getRange("A2:S")]).setBackground("#fff9c4").whenFormulaSatisfied('=AND($J2="Согласовано", $N2<>"Оплачено", $B2>=TODAY())').build());
  rules.push(SpreadsheetApp.newConditionalFormatRule().setRanges([sheet.getRange("A2:S")]).setBackground("#c8e6c9").whenFormulaSatisfied('=$N2="Оплачено"').build());
  rules.push(SpreadsheetApp.newConditionalFormatRule().setRanges([sheet.getRange("A2:S")]).setBackground("#bbdefb").whenFormulaSatisfied('=$J2="На согласовании"').build());
  sheet.setConditionalFormatRules(rules);
}

function refreshColors() { setupConditionalFormatting(); showAlert('🎨 Цвета обновлены'); }

// === ВЫПАДАЮЩИЕ СПИСКИ ===
function setupDataValidation(sheet) {
  try {
    var lastRow = 2000;
    sheet.getRange("J2:J" + lastRow).setDataValidation(SpreadsheetApp.newDataValidation().requireValueInList(["На согласовании", "Согласовано", "Отклонено"], true).build());
    sheet.getRange("N2:N" + lastRow).setDataValidation(SpreadsheetApp.newDataValidation().requireValueInList(["Не оплачено", "Оплачено", "Частично"], true).build());
    sheet.getRange("H2:H" + lastRow).setDataValidation(SpreadsheetApp.newDataValidation().requireValueInList(["RUB", "USD", "EUR", "CNY"], true).build());
  } catch (error) {}
}

// === ИНИЦИАЛИЗАЦИЯ СИСТЕМЫ ===
function initializeSystem() {
  var ss = SpreadsheetApp.getActiveSpreadsheet();
  var sheet = ss.getSheetByName(НАСТРОЙКИ.листРеестр);
  if (!sheet) sheet = ss.insertSheet(НАСТРОЙКИ.листРеестр);
  if (sheet.getLastRow() == 0) initializeNewSheet(sheet);
  sheet.hideColumns(17);
  setupDataValidation(sheet);
  setupConditionalFormatting();
  refreshCurrencyRates();
  createOrUpdateDashboard();
  removeAllProtectionsCompletely();
  showAlert('✅ Система инициализирована', 'Нажмите "🔐 АВТОРИЗАЦИЯ" для первого запуска');
}

// === ОТКРЫТЬ ПАПКУ ===
function openDriveFolder() {
  if (НАСТРОЙКИ.корневаяПапка) {
    var url = 'https://drive.google.com/drive/folders/' + НАСТРОЙКИ.корневаяПапка;
    var html = '<script>window.open("' + url + '"); google.script.host.close();</script>';
    SpreadsheetApp.getUi().showModalDialog(HtmlService.createHtmlOutput(html).setWidth(100).setHeight(100), 'Открытие папки');
  }
}

// === ОБРАБОТКА ИЗМЕНЕНИЙ ===
function onEdit(e) {
  var range = e.range;
  var sheet = range.getSheet();
  var row = range.getRow();
  var col = range.getColumn();
  if (sheet.getName() != НАСТРОЙКИ.листРеестр || row == 1) return;
  var userEmail = Session.getActiveUser().getEmail();
  if (col == 10) {
    sheet.getRange(row, 11).setValue(userEmail);
    sheet.getRange(row, 12).setValue(new Date());
  }
  if (col == 14 && range.getValue() == "Оплачено") {
    sheet.getRange(row, 15).setValue(new Date());
  }
  if (col == 7 || col == 8) {
    Utilities.sleep(100);
    addRubFormula(row);
  }
}
